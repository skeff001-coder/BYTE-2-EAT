#!/usr/bin/env bash

WORKSPACE="$CM_BUILD_DIR/$XCODE_WORKSPACE"
ARCHIVE_PATH="$CM_BUILD_DIR/build/ios/xcarchive/App.xcarchive"
IPA_DIR="$CM_BUILD_DIR/build/ios/ipa"

# Find the installed provisioning profile and extract its UUID
PROFILE_PATH=$(ls ~/Library/MobileDevice/Provisioning\ Profiles/*.mobileprovision 2>/dev/null | head -1)
echo "Profile path: $PROFILE_PATH"
PROFILE_UUID=$(security cms -D -i "$PROFILE_PATH" | python3 -c "import plistlib,sys; d=plistlib.loads(sys.stdin.buffer.read()); print(d['UUID'])")
echo "Profile UUID: $PROFILE_UUID"

# Archive
xcodebuild \
  -workspace "$WORKSPACE" \
  -scheme "$XCODE_SCHEME" \
  -config "Release" \
  -archivePath "$ARCHIVE_PATH" \
  CODE_SIGN_STYLE=Manual \
  DEVELOPMENT_TEAM="J6N9GAHK44" \
  PROVISIONING_PROFILE="$PROFILE_UUID" \
  COMPILER_INDEX_STORE_ENABLE=NO \
  archive

# Export IPA
mkdir -p "$IPA_DIR"
python3 -c "import plistlib; plistlib.dump({'method':'app-store','teamID':'J6N9GAHK44'}, open('/tmp/exportOptions.plist','wb'))"
xcodebuild \
  -exportArchive \
  -archivePath "$ARCHIVE_PATH" \
  -exportPath "$IPA_DIR" \
  -exportOptionsPlist /tmp/exportOptions.plist
